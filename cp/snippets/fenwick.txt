class FenwickTree {
public:
    vector<int> bit;
    int N;

    FenwickTree(int size) {
        N = size;
        bit.resize(N + 1, 0);
    }

    void update(int k, int val) {
        while (k <= N) {
            bit[k] += val;
            k += (k & (-k));
        }
    }

    int query(int k) {
        int res = 0;
        while (k > 0) {
            res += bit[k];
            k -= (k & (-k));
        }
        return res;
    }
};
